cmake_minimum_required(VERSION 3.22)
project(VoxTuneAutotune VERSION 1.0.0)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.4 GIT_SHALLOW TRUE)
FetchContent_Declare(SoundTouch GIT_REPOSITORY https://codeberg.org/soundtouch/soundtouch.git GIT_TAG 2.3.3 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
FetchContent_Populate(SoundTouch)
juce_add_plugin(VoxTuneAutotune COMPANY_NAME "VoxTune" COMPANY_WEBSITE "https://vox-tune.com" PLUGIN_MANUFACTURER_CODE Vxtn PLUGIN_CODE Vtat FORMATS VST3 AU Standalone PRODUCT_NAME "VoxTune Autotune" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE EDITOR_WANTS_KEYBOARD_FOCUS FALSE VST3_CATEGORIES "Fx" "Pitch Shift" COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(VoxTuneAutotune PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp Source/DSP/PitchDetector.cpp Source/DSP/ScaleEngine.cpp ${soundtouch_SOURCE_DIR}/source/SoundTouch/AAFilter.cpp ${soundtouch_SOURCE_DIR}/source/SoundTouch/FIFOSampleBuffer.cpp ${soundtouch_SOURCE_DIR}/source/SoundTouch/FIRFilter.cpp ${soundtouch_SOURCE_DIR}/source/SoundTouch/cpu_detect_x86.cpp ${soundtouch_SOURCE_DIR}/source/SoundTouch/InterpolateCubic.cpp ${soundtouch_SOURCE_DIR}/source/SoundTouch/InterpolateLinear.cpp ${soundtouch_SOURCE_DIR}/source/SoundTouch/InterpolateShannon.cpp ${soundtouch_SOURCE_DIR}/source/SoundTouch/RateTransposer.cpp ${soundtouch_SOURCE_DIR}/source/SoundTouch/SoundTouch.cpp ${soundtouch_SOURCE_DIR}/source/SoundTouch/TDStretch.cpp)
target_include_directories(VoxTuneAutotune PRIVATE Source ${soundtouch_SOURCE_DIR}/include ${soundtouch_SOURCE_DIR}/source/SoundTouch)
target_compile_definitions(VoxTuneAutotune PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 JUCE_DISPLAY_SPLASH_SCREEN=0 SOUNDTOUCH_PREVENT_COMPILER_OPTIMIZATIONS=1)
target_link_libraries(VoxTuneAutotune PRIVATE juce::juce_audio_utils juce::juce_dsp juce::juce_graphics PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_lto_flags juce::juce_recommended_warning_flags)
